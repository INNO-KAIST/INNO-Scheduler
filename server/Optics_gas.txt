/* =====================================================================
 * Weekly Synthesis Scheduler - 핵심 공유 로직 (주차 계산 + 결정론적 추첨)
 * 이 파일은 index.html 에 그대로 인라인되며, Node 단위테스트에서도 사용됨
 * ===================================================================== */

const KST_OFFSET_MS = 9 * 3600 * 1000; // Asia/Seoul, DST 없음
const DAY_MS = 86400000;

/* ---- 슬롯 정의 (canonical order) ----
 * Day   : 월~목 06:00-18:00  → D0~D3
 * Night : 월~목 18:00-익일06 → N0~N3
 * Weekend: 금/토/일 전일     → W0~W2
 * 배치: 1~4  → cellKey = "D0-1" 형태 (총 44개)
 */
const SLOT_DEFS = [
  { key: 'D0', period: 'Day',     label: '월요일 06-18시',        day: '월' },
  { key: 'D1', period: 'Day',     label: '화요일 06-18시',        day: '화' },
  { key: 'D2', period: 'Day',     label: '수요일 06-18시',        day: '수' },
  { key: 'D3', period: 'Day',     label: '목요일 06-18시',        day: '목' },
  { key: 'N0', period: 'Night',   label: '월요일 18시-화 06시',   day: '월' },
  { key: 'N1', period: 'Night',   label: '화요일 18시-수 06시',   day: '화' },
  { key: 'N2', period: 'Night',   label: '수요일 18시-목 06시',   day: '수' },
  { key: 'N3', period: 'Night',   label: '목요일 18시-금 06시',   day: '목' },
  { key: 'W0', period: 'Weekend', label: '금요일 (전일)',          day: '금' },
  { key: 'W1', period: 'Weekend', label: '토요일 (전일)',          day: '토' },
  { key: 'W2', period: 'Weekend', label: '일요일 (전일)',          day: '일' },
];
const SLOT_ORDER = SLOT_DEFS.map(s => s.key);
const BATCHES = [1, 2, 3, 4];
const ALL_CELL_KEYS = SLOT_ORDER.flatMap(s => BATCHES.map(b => `${s}-${b}`));
const TOKENS_PER_WEEK = 4;
const DAY_CAP = 2; // 1주 Day 시간대 1인 최대 배치 수

function isDaySlot(slotKey) { return slotKey[0] === 'D'; }
function parseCell(cellKey) {
  const [slot, b] = cellKey.split('-');
  return { slot, batch: parseInt(b, 10) };
}

/* ---- 주차 계산 (월요일 시작, KST 고정) ---- */
function kstDayNum(ms) { return Math.floor((ms + KST_OFFSET_MS) / DAY_MS); }
function kstDow(ms) { return (kstDayNum(ms) + 3) % 7; } // 0=월 ... 6=일 (1970-01-01=목 검증됨)
// 주 인덱스(월요일 시작). 월요일의 dayNum ≡ 4 (mod 7) 이므로 +3 보정 후 7로 나눠 항상 '정수' 반환
function weekIndexOf(ms) { return Math.floor((kstDayNum(ms) + 3) / 7); }
function weekStartMs(weekIdx) { return (weekIdx * 7 - 3) * DAY_MS - KST_OFFSET_MS; } // 해당 주 월요일 00:00 KST의 UTC ms
function fmtDate(ms) {
  const d = new Date(ms + KST_OFFSET_MS);
  return `${d.getUTCMonth() + 1}/${d.getUTCDate()}`;
}
function weekLabel(weekIdx) {
  const s = weekStartMs(weekIdx);
  return `${fmtDate(s)}(월) ~ ${fmtDate(s + 6 * DAY_MS)}(일)`;
}
/* 투표 주(현재 주)의 국면 판정. targetWeek = votingWeek + 1 */
function phaseOf(nowMs) {
  const votingWeek = weekIndexOf(nowMs);
  const targetWeek = votingWeek + 1;
  const monday = weekStartMs(votingWeek);
  const voteClose = monday + 3 * DAY_MS;            // 목요일 00:00 (= 수요일 24시)
  const announce = monday + 3 * DAY_MS + 12 * 3600 * 1000; // 목요일 12:00
  let phase;
  if (nowMs < voteClose) phase = 'VOTING';
  else if (nowMs < announce) phase = 'WAITING';
  else phase = 'BOOKING';
  return { votingWeek, targetWeek, monday, voteClose, announce, phase };
}

/* ---- 결정론적 시드 및 PRNG ---- */
/* 투표 스냅샷의 canonical JSON: memberId 정렬, 각 멤버의 cell 목록 정렬 */
function canonicalVotesJson(votesByMember) {
  const ids = Object.keys(votesByMember).sort();
  const obj = {};
  for (const id of ids) {
    obj[id] = [...votesByMember[id]].sort();
  }
  return JSON.stringify(obj);
}

/* SHA-256 (의존성 없는 순수 JS 구현; 브라우저/Node 동일 결과) */
function sha256Hex(str) {
  const K = [0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,
    0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,
    0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,
    0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,
    0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,
    0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,0xd192e819,0xd6990624,0xf40e3585,0x106aa070,
    0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,
    0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2];
  let H = [0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19];
  const rr = (x, n) => (x >>> n) | (x << (32 - n));
  // UTF-8 인코딩
  const bytes = [];
  for (let i = 0; i < str.length; i++) {
    let c = str.codePointAt(i);
    if (c > 0xffff) i++;
    if (c < 0x80) bytes.push(c);
    else if (c < 0x800) bytes.push(0xc0 | (c >> 6), 0x80 | (c & 63));
    else if (c < 0x10000) bytes.push(0xe0 | (c >> 12), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
    else bytes.push(0xf0 | (c >> 18), 0x80 | ((c >> 12) & 63), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
  }
  const bitLen = bytes.length * 8;
  bytes.push(0x80);
  while (bytes.length % 64 !== 56) bytes.push(0);
  for (let i = 7; i >= 0; i--) bytes.push((i >= 4 ? 0 : (bitLen / Math.pow(2, i * 8))) & 0xff);
  // 상위 32비트는 실용상 0 (2^32비트 미만 입력)
  for (let off = 0; off < bytes.length; off += 64) {
    const w = new Array(64);
    for (let i = 0; i < 16; i++)
      w[i] = (bytes[off + 4 * i] << 24) | (bytes[off + 4 * i + 1] << 16) | (bytes[off + 4 * i + 2] << 8) | bytes[off + 4 * i + 3];
    for (let i = 16; i < 64; i++) {
      const s0 = rr(w[i - 15], 7) ^ rr(w[i - 15], 18) ^ (w[i - 15] >>> 3);
      const s1 = rr(w[i - 2], 17) ^ rr(w[i - 2], 19) ^ (w[i - 2] >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) | 0;
    }
    let [a, b, c, d, e, f, g, h] = H;
    for (let i = 0; i < 64; i++) {
      const S1 = rr(e, 6) ^ rr(e, 11) ^ rr(e, 25);
      const ch = (e & f) ^ (~e & g);
      const t1 = (h + S1 + ch + K[i] + w[i]) | 0;
      const S0 = rr(a, 2) ^ rr(a, 13) ^ rr(a, 22);
      const mj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (S0 + mj) | 0;
      h = g; g = f; f = e; e = (d + t1) | 0; d = c; c = b; b = a; a = (t1 + t2) | 0;
    }
    H = H.map((x, i) => (x + [a, b, c, d, e, f, g, h][i]) | 0);
  }
  return H.map(x => (x >>> 0).toString(16).padStart(8, '0')).join('');
}

/* mulberry32 - 시드 기반 결정론적 PRNG */
function makeRng(seedHex) {
  let s = parseInt(seedHex.slice(0, 8), 16) >>> 0;
  // 시드 확장: 해시 뒷부분도 섞음
  s = (s ^ parseInt(seedHex.slice(8, 16), 16)) >>> 0;
  return function () {
    s = (s + 0x6D2B79F5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function shuffled(arr, rng) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

/* =====================================================================
 * 결정론적 추첨
 * 입력: votesByMember = { memberId: ["D0-1","D0-2","N1-3", ...] }
 * 출력: { seed, assignments: {cellKey: memberId}, priorityMembers: [...],
 *         dayCount: {memberId: n} }
 *
 * 규칙 구현:
 * (1) 동일 슬롯+배치 1인 배정, 경합 시 시드 기반 무작위 추첨
 * (2) Day 시간대 주간 1인 최대 2배치 (배정 시점 강제)
 * (3) 동일 슬롯에 2개 이상 배치 투표자(번들)가 우선권, 번들 간 경합도 무작위
 * (4) 번들은 all-or-nothing (전부 배정 또는 전부 탈락)
 * ===================================================================== */
function computeLottery(votesByMember) {
  const canonical = canonicalVotesJson(votesByMember);
  const seed = sha256Hex(canonical);
  const rng = makeRng(seed);

  const assignments = {}; // cellKey -> memberId
  const dayCount = {};    // memberId -> Day 배정 수
  const memberIds = Object.keys(votesByMember).sort();

  for (const slot of SLOT_ORDER) {
    const day = isDaySlot(slot);
    // 이 슬롯에서 멤버별 투표 배치 목록
    const bySlot = [];
    for (const m of memberIds) {
      const bs = (votesByMember[m] || [])
        .map(parseCell).filter(c => c.slot === slot).map(c => c.batch)
        .sort((a, b) => a - b);
      if (bs.length) bySlot.push({ m, bs });
    }
    // --- 1단계: 번들(같은 슬롯 2개 이상 배치) 우선, 무작위 순서 그리디 ---
    const bundles = shuffled(bySlot.filter(x => x.bs.length >= 2), rng);
    for (const { m, bs } of bundles) {
      if (day && (dayCount[m] || 0) + bs.length > DAY_CAP) continue; // 규칙(2)
      const free = bs.every(b => !assignments[`${slot}-${b}`]);
      if (free) { // 규칙(4): all-or-nothing
        for (const b of bs) assignments[`${slot}-${b}`] = m;
        if (day) dayCount[m] = (dayCount[m] || 0) + bs.length;
      }
    }
    // --- 2단계: 단일 투표자, 배치별 무작위 추첨 ---
    const singles = bySlot.filter(x => x.bs.length === 1);
    for (const b of BATCHES) {
      const key = `${slot}-${b}`;
      if (assignments[key]) continue;
      const cands = singles
        .filter(x => x.bs[0] === b)
        .filter(x => !(day && (dayCount[x.m] || 0) >= DAY_CAP))
        .map(x => x.m); // memberIds 순회로 이미 정렬됨
      if (!cands.length) continue;
      const pick = cands.length === 1 ? cands[0] : cands[Math.floor(rng() * cands.length)];
      assignments[key] = pick;
      if (day) dayCount[pick] = (dayCount[pick] || 0) + 1;
    }
  }

  // Day에 1표 이상 투자했으나 Day 배정 0인 인원 → 우선예약 대상
  const priorityMembers = memberIds.filter(m => {
    const votedDay = (votesByMember[m] || []).some(c => isDaySlot(parseCell(c).slot));
    return votedDay && !(dayCount[m] > 0);
  });

  return { seed, assignments, priorityMembers, dayCount, canonical };
}

/* 투표 유효성 검사 (클라이언트 측; 서버 rules 와 동일 기준) */
function validateVoteSet(cells) {
  if (!Array.isArray(cells)) return '형식 오류';
  if (cells.length > TOKENS_PER_WEEK) return `토큰은 주당 ${TOKENS_PER_WEEK}개까지입니다.`;
  const set = new Set(cells);
  if (set.size !== cells.length) return '동일 슬롯·배치에 중복 투표할 수 없습니다.';
  for (const c of cells) if (!ALL_CELL_KEYS.includes(c)) return `유효하지 않은 슬롯: ${c}`;
  // 단일 Day 슬롯 내 3개 이상 번들은 규칙(2)와 (4)의 조합상 절대 당첨 불가 → 사전 차단
  for (const s of SLOT_ORDER.filter(isDaySlot)) {
    const n = cells.filter(c => parseCell(c).slot === s).length;
    if (n > DAY_CAP) return `Day 슬롯(${s}) 하나에 ${n}개 배치 묶음은 Day 주간 2개 제한과 전부-또는-전무 규칙상 당첨이 불가능하여 차단됩니다.`;
  }
  return null;
}

/* =====================================================================
 * Weekly Optics Scheduler - 광학 전용 핵심 로직
 * (core.js 의 주차계산·SHA-256·PRNG 를 그대로 사용하며, 이 파일이 뒤에 연결됨)
 * ===================================================================== */

const OPT_DAYS = ['월', '화', '수', '목', '금', '토', '일']; // dayIdx 0~6
const OPT_SLOTS_PER_DAY = 48; // 30분 단위 00:00~24:00
const OPT_MAX_RANKS = 3;

function optSlotLabel(slotIdx) {
  const h = Math.floor(slotIdx / 2), m = slotIdx % 2 ? '30' : '00';
  return String(h).padStart(2, '0') + ':' + m;
}
function optDayDate(weekIdx, dayIdx) { return weekStartMs(weekIdx) + dayIdx * DAY_MS; }
function optDayLabel(weekIdx, dayIdx) { return fmtDate(optDayDate(weekIdx, dayIdx)) + '(' + OPT_DAYS[dayIdx] + ')'; }
/* '7/16(목)' 형태 + 요일 포함 날짜 표기 (하드웨어 로그용) */
/* Remark 기록: 새 항목을 맨 앞에 추가하고 최대 max개만 유지(오래된 것 자동 삭제) */
function optPushRemark(entries, entry, max) {
  const cap = max || 10;
  const list = [entry].concat(Array.isArray(entries) ? entries : []);
  return list.slice(0, cap);
}

function optFmtDateDow(ms) {
  const dow = OPT_DAYS[(kstDayNum(ms) + 3) % 7];
  const d = new Date(ms + KST_OFFSET_MS);
  return `${d.getUTCFullYear()}.${String(d.getUTCMonth() + 1).padStart(2, '0')}.${String(d.getUTCDate()).padStart(2, '0')}(${dow}) ${String(d.getUTCHours()).padStart(2, '0')}:${String(d.getUTCMinutes()).padStart(2, '0')}`;
}

/* =====================================================================
 * 관리자 전용 블록 (Repair / 외부인 방문측정)
 * ---------------------------------------------------------------------
 * 30분 슬롯의 '소유자' 값으로 멤버ID 대신 특수 문자열을 사용한다.
 *   형식: '!' + kind + ('|' + memo)      예) '!repair', '!visit|A대 김OO'
 * Firestore 자동생성 ID는 영숫자뿐이므로 '!' 접두는 멤버ID와 절대 충돌하지 않는다.
 * 이 값은 optEffectiveSchedule 의 '최상위 레이어'로 적용되어 추첨 배정·자유
 * 조율 결과를 모두 덮는다 → 당첨자는 관리자 블록을 제외한 나머지 시간만 사용.
 * ===================================================================== */
const OPT_ADMIN_MARK = '!';
const OPT_ADMIN_KINDS = [
  { code: 'repair', label: 'Repair' },
  { code: 'visit', label: '외부인 방문측정' },
];
function optIsAdminOwner(v) { return typeof v === 'string' && v.charAt(0) === OPT_ADMIN_MARK; }
/* 메모 정제: 캘린더 태그([WOS|…]) 및 값 구분자와 충돌하는 문자 제거 + 길이 제한 */
function optAdminMemoClean(memo) {
  return String(memo == null ? '' : memo).replace(/[\[\]|\r\n\t]/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 40);
}
function optAdminOwner(kind, memo) {
  const m = optAdminMemoClean(memo);
  return OPT_ADMIN_MARK + kind + (m ? '|' + m : '');
}
function optAdminParse(v) {
  if (!optIsAdminOwner(v)) return null;
  const body = v.slice(1);
  const i = body.indexOf('|');
  const code = i < 0 ? body : body.slice(0, i);
  const memo = i < 0 ? '' : body.slice(i + 1);
  const k = OPT_ADMIN_KINDS.filter(function (x) { return x.code === code; })[0];
  return { code: code, kind: k ? k.label : code, memo: memo };
}
function optAdminLabel(v) {
  const p = optAdminParse(v);
  return p ? p.kind + (p.memo ? ' · ' + p.memo : '') : '';
}
/* 관리자 블록 문서에서 요일별 슬롯 배열(48) 추출. '' 또는 미기재 = 블록 없음 */
function optAdminSlotArr(adminDocs, day) {
  const slots = ((adminDocs || {})[String(day)] || {}).slots || {};
  const arr = new Array(OPT_SLOTS_PER_DAY);
  for (let s = 0; s < OPT_SLOTS_PER_DAY; s++) {
    const v = slots[String(s)];
    arr[s] = (v === undefined || v === null || v === '') ? null : v;
  }
  return arr;
}
/* 요일의 관리자 블록을 연속 구간으로 병합 → [{start, end, owner}] */
function optAdminBlocksOfDay(adminDocs, day) { return optMergeBlocks(optAdminSlotArr(adminDocs, day)); }
/* 요일 중 관리자 블록이 덮지 않은 30분 슬롯 수 (0이면 당첨자 가용 시간 없음) */
function optFreeSlotCount(adminDocs, day) {
  return optAdminSlotArr(adminDocs, day).filter(function (v) { return v === null; }).length;
}
/* 슬롯 구간 라벨: end 는 미포함 인덱스 */
function optRangeLabel(start, end) {
  return optSlotLabel(start) + '~' + (end >= OPT_SLOTS_PER_DAY ? '24:00' : optSlotLabel(end));
}

/* =====================================================================
 * 하루 4분할 (Quarter) — 투표 단위
 * Q1 새벽 00-06 / Q2 오전 06-12 / Q3 오후 12-18 / Q4 저녁 18-24
 * 30분 슬롯 기준으로 각 분기는 12칸을 차지한다 (12 × 30분 = 6시간).
 * ===================================================================== */
const OPT_QUARTERS = [
  { q: 1, key: 'Q1', label: '새벽', range: '00-06' },
  { q: 2, key: 'Q2', label: '오전', range: '06-12' },
  { q: 3, key: 'Q3', label: '오후', range: '12-18' },
  { q: 4, key: 'Q4', label: '저녁', range: '18-24' },
];
const OPT_SLOTS_PER_QUARTER = OPT_SLOTS_PER_DAY / 4; // 12
const OPT_ALL_QUARTERS = [1, 2, 3, 4];

function optQuarterLabel(q) {
  const x = OPT_QUARTERS.filter(function (o) { return o.q === q; })[0];
  return x ? x.key + ' ' + x.label + '(' + x.range + ')' : 'Q' + q;
}
/* 분기 목록 → '00-12시' 처럼 연속 구간을 합친 짧은 표기 */
function optQuartersLabel(qs) {
  const sorted = (qs || []).slice().sort(function (a, b) { return a - b; });
  if (!sorted.length) return '';
  if (sorted.length === 4) return '종일';
  const parts = [];
  let st = sorted[0], prev = sorted[0];
  for (let i = 1; i <= sorted.length; i++) {
    const cur = sorted[i];
    if (cur === prev + 1) { prev = cur; continue; }
    parts.push(String((st - 1) * 6).padStart(2, '0') + '-' + String(prev * 6).padStart(2, '0'));
    st = cur; prev = cur;
  }
  return parts.join(', ') + '시';
}

/* ---- 투표 항목 정규화 ----
 * 신형: {d: 요일idx, q: [1,2]}   구형(하위호환): 요일idx 숫자 하나 = 하루 전체
 * 반환 {d, q(정렬·중복제거된 1~4)} 또는 null */
function optNormRank(v) {
  if (v === undefined || v === null) return null;
  if (typeof v === 'number') {
    return Number.isInteger(v) && v >= 0 && v <= 6 ? { d: v, q: OPT_ALL_QUARTERS.slice() } : null;
  }
  if (typeof v !== 'object' || !Number.isInteger(v.d) || v.d < 0 || v.d > 6) return null;
  let q = Array.isArray(v.q) ? v.q : OPT_ALL_QUARTERS;
  q = q.filter(function (x) { return Number.isInteger(x) && x >= 1 && x <= 4; });
  q = q.filter(function (x, i) { return q.indexOf(x) === i; }).sort(function (a, b) { return a - b; });
  if (!q.length) return null;
  return { d: v.d, q: q };
}
/* 투표 묶음 {r1,r2,r3} 을 전부 신형 {d,q} 로 정규화 (유효하지 않은 항목은 제거).
 * 저장 직전과 편집 상태 초기화 시 반드시 통과시킬 것 — 구버전에서 저장된 숫자 형식이
 * 그대로 다시 전송되면 서버 규칙(신형 형식 검사)에 걸려 'Missing or insufficient permissions' 가 난다. */
function optNormRanks(ranks) {
  const out = {};
  ['r1', 'r2', 'r3'].forEach(function (k) {
    const n = optNormRank((ranks || {})[k]);
    if (n) out[k] = { d: n.d, q: n.q };
  });
  return out;
}
/* 분기 소유 맵 {"1":mid,…} → 48슬롯 배열 */
function optQuarterSlots(qmap) {
  const arr = new Array(OPT_SLOTS_PER_DAY).fill(null);
  for (let k = 1; k <= 4; k++) {
    const mid = (qmap || {})[String(k)];
    if (!mid) continue;
    for (let s = (k - 1) * OPT_SLOTS_PER_QUARTER; s < k * OPT_SLOTS_PER_QUARTER; s++) arr[s] = mid;
  }
  return arr;
}
/* 결과 문서 → 요일별 48슬롯 '기본 배정' 배열.
 * quarterAssign 이 있으면 분기 단위로, 없으면(구버전 결과) dayAssign 으로 하루 전체. */
function optBaseSlots(results) {
  const r = results || {};
  // dayAssign 맵을 그대로 넘겨준 옛 호출도 계속 지원 (키가 "0"~"6" 뿐이면 dayAssign 으로 간주)
  const isRawDayAssign = r.dayAssign === undefined && r.quarterAssign === undefined;
  const da = isRawDayAssign ? r : (r.dayAssign || {});
  const qa = isRawDayAssign ? null : r.quarterAssign;
  const out = {};
  for (let day = 0; day < 7; day++) {
    if (qa && qa[String(day)]) { out[day] = optQuarterSlots(qa[String(day)]); continue; }
    const mid = da[String(day)] || null;
    out[day] = new Array(OPT_SLOTS_PER_DAY).fill(mid);
  }
  return out;
}


/* ---- 투표 canonical 직렬화 ----
 * votesByMember = { memberId: {r1: dayIdx, r2?: dayIdx, r3?: dayIdx} } */
function optCanonicalVotesJson(votesByMember) {
  const ids = Object.keys(votesByMember).sort();
  const obj = {};
  for (const id of ids) {
    const v = votesByMember[id] || {};
    const o = {};
    for (const k of ['r1', 'r2', 'r3']) {
      const n = optNormRank(v[k]);
      if (!n) continue;
      // ★ 하위호환: '종일' 지원은 구버전과 똑같이 요일 숫자 하나로 직렬화한다.
      //   그래야 시간대 기능 도입 전에 발표된 결과(및 과거 주차)의 시드가 그대로 재현되어
      //   전원 재검증에서 '검증 불일치' 오탐이 나지 않는다.
      //   시간대를 일부만 고른 투표가 하나라도 있으면 그 항목만 {d,q} 로 들어간다.
      o[k] = n.q.length === 4 ? n.d : { d: n.d, q: n.q };
    }
    obj[id] = o;
  }
  return JSON.stringify(obj);
}

/* ---- 투표 유효성 (클라이언트; 서버 rules 와 동일 기준) ---- */
function optValidateVote(ranks) {
  const keys = Object.keys(ranks || {});
  for (const k of keys) if (!['r1', 'r2', 'r3'].includes(k)) return '형식 오류';
  const days = [];
  for (const k of keys) {
    const n = optNormRank(ranks[k]);
    if (!n) return '유효하지 않은 요일 또는 시간대';
    days.push(n.d);
  }
  if (new Set(days).size !== days.length) return '같은 요일을 중복 순위로 투표할 수 없습니다.';
  // 순위는 1부터 연속 (r2가 있으면 r1도, r3가 있으면 r2도)
  if ((ranks.r2 !== undefined && ranks.r1 === undefined) || (ranks.r3 !== undefined && ranks.r2 === undefined))
    return '순위는 1순위부터 순서대로 지정해야 합니다.';
  return null;
}

/* =====================================================================
 * 결정론적 요일 배정
 * 정렬 기준(중요, 이 순서대로):
 *   ① 현재까지 당첨 수가 적은 사람 (무당첨자 절대 우선)
 *   ② 해당 요일에 대한 순위가 높은(숫자가 작은) 사람
 *   ③ 전체 투표 수가 적은 사람 (더 간절함)
 *   ④ 시드 기반 무작위 (시드 = 마감된 전체 투표의 SHA-256 → 조작 불가·전원 검증 가능)
 * 요일 처리 순서는 월→일 고정 (당첨 수 갱신이 다음 요일에 반영됨)
 * 한 사람이 여러 요일에 당첨될 수 있음 (단 ①에 의해 무당첨자가 항상 우선)
 * ===================================================================== */
function computeOpticsLottery(votesByMember) {
  const canonical = optCanonicalVotesJson(votesByMember);
  const seed = sha256Hex(canonical);
  const rng = makeRng(seed);

  const memberIds = Object.keys(votesByMember).sort();
  const voteCount = {};       // mid -> 투표한 순위 수 (1~3)
  const rankOf = {};          // mid -> {dayIdx: rank}
  const quartOf = {};         // mid -> {dayIdx: [분기…]}
  for (const mid of memberIds) {
    const v = votesByMember[mid] || {};
    rankOf[mid] = {}; quartOf[mid] = {};
    let n = 0;
    [['r1', 1], ['r2', 2], ['r3', 3]].forEach(function (pair) {
      const nm = optNormRank(v[pair[0]]);
      if (!nm) return;
      rankOf[mid][nm.d] = pair[1];
      quartOf[mid][nm.d] = nm.q;
      n++;
    });
    voteCount[mid] = n;
  }

  const wins = {};            // mid -> 시간을 받은 요일 수
  const dayAssign = {};       // dayIdx(string) -> 대표 memberId (표시·하위호환용)
  const quarterAssign = {};   // dayIdx(string) -> { "1": mid, … }  실제 소유
  const dayContenders = {};   // dayIdx -> [{mid, rank, q}] (결과 표시용)

  /* 후보 목록에서 ①~④ 기준 최소 튜플을 뽑는다 (동률이면 시드 난수) */
  function pick(cands) {
    let best = null;
    for (const c of cands) {
      const w = wins[c.mid] || 0;
      if (!best || w < best.w
          || (w === best.w && (c.rank < best.rank
          || (c.rank === best.rank && voteCount[c.mid] < best.v)))) {
        best = { mid: c.mid, w: w, rank: c.rank, v: voteCount[c.mid] };
      }
    }
    const tied = cands.filter(function (c) {
      return (wins[c.mid] || 0) === best.w && c.rank === best.rank && voteCount[c.mid] === best.v;
    });
    return tied.length === 1 ? tied[0] : tied[Math.floor(rng() * tied.length)];
  }

  for (let day = 0; day < 7; day++) {
    const cands = memberIds
      .filter(function (mid) { return rankOf[mid][day] !== undefined; })
      .map(function (mid) { return { mid: mid, rank: rankOf[mid][day], q: quartOf[mid][day] }; });
    dayContenders[day] = cands.map(function (c) { return { mid: c.mid, rank: c.rank, q: c.q.slice() }; });
    if (!cands.length) continue;

    const owner = {};                       // 분기 -> mid
    const taken = {};                       // 이미 배정된 mid
    const remaining = cands.slice();

    // 1단계: 대표
    const first = pick(remaining);
    first.q.forEach(function (q) { owner[q] = first.mid; });
    taken[first.mid] = true;
    wins[first.mid] = (wins[first.mid] || 0) + 1;
    dayAssign[String(day)] = first.mid;

    // 2단계: 남은 분기에 '통째로' 들어갈 수 있는 사람을 반복해서 채움
    for (;;) {
      const fits = cands.filter(function (c) {
        if (taken[c.mid]) return false;
        return c.q.every(function (q) { return owner[q] === undefined; });
      });
      if (!fits.length) break;
      const nx = pick(fits);
      nx.q.forEach(function (q) { owner[q] = nx.mid; });
      taken[nx.mid] = true;
      wins[nx.mid] = (wins[nx.mid] || 0) + 1;
    }

    const qmap = {};
    for (let k = 1; k <= 4; k++) if (owner[k]) qmap[String(k)] = owner[k];
    quarterAssign[String(day)] = qmap;
  }

  // 1표 이상 투표했으나 어떤 시간대도 받지 못한 인원 (참고 표시용)
  const unassignedMembers = memberIds.filter(function (mid) { return voteCount[mid] > 0 && !(wins[mid] > 0); });

  return { seed, dayAssign, quarterAssign, unassignedMembers, canonical, dayContenders };
}

/* ---- 유효 스케줄 계산: 기본 요일 배정 + 30분 단위 수정본(override) + 관리자 블록 ----
 * scheduleDocs: {dayIdx(string): {slots: {slotIdx(string): memberId|""}}}
 * adminDocs   : {dayIdx(string): {slots: {slotIdx(string): '!kind|memo'|""}}}  (선택)
 * 우선순위(높은 쪽이 이김): 관리자 블록 > 자유 조율 override > 시간대(분기) 기본 배정
 * 첫 인자는 결과 문서 전체(quarterAssign 사용). 구형 dayAssign 맵도 받는다(optBaseSlots).
 * 반환: day -> [48] 배열 (memberId | '!kind…' | null)                          */
function optEffectiveSchedule(resultsOrDayAssign, scheduleDocs, adminDocs) {
  const baseSlots = optBaseSlots(resultsOrDayAssign);   // 분기 배정 → 48슬롯 (구형 dayAssign 도 수용)
  const eff = {};
  for (let day = 0; day < 7; day++) {
    const base = baseSlots[day];
    const slots = ((scheduleDocs || {})[String(day)] || {}).slots || {};
    const adm = optAdminSlotArr(adminDocs, day);
    const arr = new Array(OPT_SLOTS_PER_DAY);
    for (let s = 0; s < OPT_SLOTS_PER_DAY; s++) {
      if (adm[s] !== null) { arr[s] = adm[s]; continue; }   // 관리자 블록이 최우선
      const ov = slots[String(s)];
      arr[s] = ov === undefined ? base[s] : (ov === '' ? null : ov);
    }
    eff[day] = arr;
  }
  return eff;
}

/* 슬롯 배열 → 연속 블록 병합 [{start, end(미포함), owner}] (캘린더/표시용) */
function optMergeBlocks(slotArr) {
  const blocks = [];
  let cur = null;
  for (let s = 0; s < slotArr.length; s++) {
    const o = slotArr[s];
    if (cur && cur.owner === o) { cur.end = s + 1; continue; }
    if (cur && cur.owner !== null) blocks.push(cur);
    cur = o !== null ? { start: s, end: s + 1, owner: o } : null;
    if (o === null) cur = null;
  }
  if (cur && cur.owner !== null) blocks.push(cur);
  return blocks;
}

function fromFs_(v) {

  if (v.integerValue !== undefined) return parseInt(v.integerValue, 10);

  if (v.booleanValue !== undefined) return v.booleanValue;
  if (v.nullValue !== undefined) return null;

  if (typeof x === 'boolean') return { booleanValue: x };

  if (Array.isArray(x)) return { arrayValue: { values: x.map(toFs_) } };
  if (typeof x === 'object') { const f = {}; for (const k in x) f[k] = toFs_(x[k]); return { mapValue: { fields: f } }; }

    if (r.j && r.j.error) throw new Error('LIST ' + collection + ': ' + r.j.error.message);
    ((r.j && r.j.documents) || []).forEach(function (d) {
      docs.push(Object.assign({ id: d.name.split('/').pop() }, decodeFields_(d.fields || {})));
    });
    pageToken = (r.j && r.j.nextPageToken) || '';
  } while (pageToken);
  return docs;
}
/* 하드웨어 로그 증분 조회: tsMs > since 인 것만.
 * 이전에는 매분 컬렉션 전체를 fsList_ 로 읽어(문서 N개 × 1440분/일) Firestore 무료
 * 읽기 한도(5만/일)를 단독으로 초과시켰다. 신규가 없으면 이 쿼리는 1 read. */
function fsQueryHwSince_(token, since) {
  const r = http_(FS_BASE_() + ':runQuery', {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ structuredQuery: {
      from: [{ collectionId: 'opticsHardwareLog' }],
      where: { fieldFilter: { field: { fieldPath: 'tsMs' }, op: 'GREATER_THAN', value: { integerValue: String(since) } } },
      orderBy: [{ field: { fieldPath: 'tsMs' }, direction: 'ASCENDING' }],
      limit: 30,
    } }),
  });
  if (r.j && r.j.error) throw new Error('QUERY opticsHardwareLog: ' + r.j.error.message);
  return (Array.isArray(r.j) ? r.j : []).filter(function (x) { return x.document; })
    .map(function (x) { return Object.assign({ id: x.document.name.split('/').pop() }, decodeFields_(x.document.fields || {})); });
}
function fsQueryByWeek_(token, collection, week) {
  const r = http_(FS_BASE_() + ':runQuery', {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ structuredQuery: {
      from: [{ collectionId: collection }],
      where: { fieldFilter: { field: { fieldPath: 'week' }, op: 'EQUAL', value: { integerValue: String(week) } } },
    } }),
  });
  if (r.j && r.j.error) throw new Error('QUERY ' + collection + ': ' + r.j.error.message);
  return (Array.isArray(r.j) ? r.j : []).filter(function (x) { return x.document; })
    .map(function (x) { return Object.assign({ id: x.document.name.split('/').pop() }, decodeFields_(x.document.fields || {})); });
}
/* 문서 1건 생성-또는-갱신(PATCH + updateMask). 없으면 만들고 있으면 해당 필드만 덮어쓴다. */
function fsSet_(token, path, obj) {
  const mask = Object.keys(obj).map(function (k) { return 'updateMask.fieldPaths=' + encodeURIComponent(k); }).join('&');
  const r = http_(FS_BASE_() + '/' + path + '?' + mask, {
    method: 'patch', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ fields: encodeFields_(obj) }),
  });
  if (r.j && r.j.error) throw new Error('SET ' + path + ': ' + r.j.error.message);
  return true;
}

/* ── 하트비트 ───────────────────────────────────────────────────────────
 * 정상 동작할 때마다 syncMeta/wos_bot.okMs 를 갱신한다. 웹앱은 이 값이
 * 60분 이상 낡으면 화면 상단에 "구글 캘린더 동기화가 멈춰 있습니다" 배너를 띄운다.
 * 30분 기록 + 60분 경고 = "한 번 걸러도 괜찮고, 두 번 연속 놓치면 경고" 구조.
 *
 * 왜 필요한가 — 2026-10-02, 이 GAS 프로젝트가 휴지통에 들어가면서 5일간 멈췄다.
 * 휴지통에 있는 스크립트는 Google 이 아예 호출하지 않으므로 실행 오류도,
 * 실패 요약 메일도 남지 않는다. 트리거는 '등록됨 · 오류율 0%' 로 멀쩡해 보였고
 * 아무도 알아채지 못한 채 구글 캘린더만 조용히 낡아갔다. 이 하트비트는 그
 * '조용한 죽음'을 사람이 볼 수 있게 만드는 장치다.
 *
 * 쓰기 비용: 1분마다가 아니라 최대 30분 간격 = 봇당 하루 약 48회. 무료 쓰기 한도
 * (2만/일)에 비하면 무시할 수준이다. 실패해도 본 작업을 막지 않는다. */
const HEARTBEAT_MIN = 30;

function heartbeat_() {
  try {
    const props = PropertiesService.getScriptProperties();
    const now = Date.now();
    const last = parseInt(props.getProperty('wos_hb_at') || '0', 10);
    if (now - last < HEARTBEAT_MIN * 60000) return;
    fsSet_(anonToken_(), 'syncMeta/wos_bot', { okMs: now });
    props.setProperty('wos_hb_at', String(now));
  } catch (e) {
    Logger.log('하트비트 기록 실패(무시됨): ' + e.message);
  }
}

function fsCreate_(token, collection, docId, obj) {
  const r = http_(FS_BASE_() + '/' + collection + '?documentId=' + encodeURIComponent(docId), {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ fields: encodeFields_(obj) }),
  });
  if (r.j && r.j.error) {
    if (r.code === 409) return false;
    throw new Error('CREATE ' + collection + '/' + docId + ': ' + r.j.error.message);
  }
  return true;
}
function postToSlack_(text) {
  if (!GAS_CONFIG.slackWebhookUrl) throw new Error('slackWebhookUrl이 비어 있습니다.');
  const r = http_(GAS_CONFIG.slackWebhookUrl, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ text: text }) });
  if (r.code !== 200) throw new Error('Slack 전송 실패(' + r.code + '): ' + r.text.slice(0, 200));
}

/* ---------------- 영어 출력 (Slack·캘린더 공용 문구) ----------------
 * 2026-11-11: Slack 공지와 캘린더 제목은 이름만 빼고 영어. 공유 코어(optDayLabel·

const EN_DOW_ = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
const EN_DOW_OF_ = { '월': 'Mon', '화': 'Tue', '수': 'Wed', '목': 'Thu', '금': 'Fri', '토': 'Sat', '일': 'Sun' };
function enDowParen_(s) { return String(s).replace(/\(([월화수목금토일])\)/g, function (m, d) { return '(' + EN_DOW_OF_[d] + ')'; }); }
function weekLabelEn_(w) { return enDowParen_(weekLabel(w)); }

function optQuartersLabelEn_(qs) { const t = optQuartersLabel(qs); return t === '종일' ? 'all day' : t.replace(/시$/, 'h'); }
const OPT_ADMIN_KIND_EN_ = { repair: 'Repair', visit: 'External visit' };
function optAdminLabelEn_(v) {
  const p = optAdminParse(v);
  return p ? (OPT_ADMIN_KIND_EN_[p.code] || p.kind) + (p.memo ? ' · ' + p.memo : '') : '';
}
function leftTextEn_(leftMs) {
  const h = Math.floor(leftMs / 3600000);
  return h >= 24 ? Math.floor(h / 24) + 'd ' + (h % 24) + 'h' : h + 'h ' + Math.floor(leftMs % 3600000 / 60000) + 'm';
}
/* 하드웨어 변경 요약은 웹앱이 한국어 고정 문구로 저장한다 → 고정 문구만 영어로(사용자가 쓴 값은 그대로) */
const HW_SECTION_EN_ = { '기타 요소': 'Other items' };
const HW_FIELD_EN_ = { '파장범위': 'Wavelength range', '파장': 'Wavelength', '컨트롤러': 'Controller', '번호': 'No.', '품목': 'Item', '위치': 'Location' };
function hwSummaryEn_(text) {
  return String(text || '').split('\n').map(function (line) {
    return line
      .replace(/: 제거됨$/, ': removed')
      .replace(/: 신규 추가 \(/, ': newly added (')
      .replace(/(: |, |\()(파장범위|파장|컨트롤러|번호|품목|위치)(?=[ =])/g, function (m, p, f) { return p + HW_FIELD_EN_[f]; })
      .replace(/([ =]|→ )(연결|해제)(?=$|,|\)| →)/g, function (m, p, v) { return p + (v === '연결' ? 'connected' : 'disconnected'); })
      .replace(/\(없음\)/g, '(none)').replace(/\(무명\)/g, '(unnamed)');
  }).join('\n');
}

/* ---------------- 월/화/수 12:00 — 투표 리마인드 ---------------- */
function buildOptRemindMessage_(targetWeek, voteClose, opticsMembers, votedIds) {
  const nowMs = Date.now();
  const voted = opticsMembers.filter(function (m) { return votedIds[m.id]; });
  const notVoted = opticsMembers.filter(function (m) { return !votedIds[m.id] && slackIdOf_(m); });
  const leftMs = voteClose - nowMs;
  const leftStr = leftTextEn_(leftMs);
  const isLastDay = leftMs <= 12.5 * 3600000;
  const lines = [];
  lines.push('🔬 *Optics pre-vote (day ranking) reminder — for ' + weekLabelEn_(targetWeek) + '*');
  lines.push('');
  lines.push(isLastDay ? '⚠ *Voting closes tonight at midnight (Wed 24:00)!*' : 'Voting closes in *' + leftStr + '* (Wed 24:00 KST)');
  lines.push('✅ Voted: *' + voted.length + '* / ' + opticsMembers.length + ' active Optics members');
  if (notVoted.length) {
    lines.push('⏳ Not voted yet: ' + notVoted.map(function (m) { return mention_(m); }).join(' '));
  }
  lines.push('');
  lines.push('• Pick your 1st/2nd/3rd choice days (no need to fill all three; fewer votes win same-rank ties) · Announcement: Thu 12:00');

}

/* 발표용: 요일 d 의 시간대별 당첨자 → '*조형구* 06-12시 · *유원정* 12-24시 · _00-06시 비어 있음_'
 * quarterAssign 이 없는 구형 결과는 dayAssign(하루 전체)로 표시한다. */
function optDayOwnersText_(results, d, names) {
  const qa = ((results || {}).quarterAssign || {})[String(d)];
  if (!qa) {
    const mid = ((results || {}).dayAssign || {})[String(d)];
    return mid ? '*' + (names[mid] || '?') + '*' : '_Unassigned (free to use)_';
  }
  const byMid = {}, free = [];
  OPT_ALL_QUARTERS.forEach(function (q) {
    const m = qa[String(q)];
    if (m) (byMid[m] = byMid[m] || []).push(q); else free.push(q);
  });
  const mids = Object.keys(byMid).sort(function (x, y) { return byMid[x][0] - byMid[y][0]; });
  if (!mids.length) return '_Unassigned (free to use)_';
  const parts = mids.map(function (m) { return '*' + (names[m] || '?') + '* ' + optQuartersLabelEn_(byMid[m]); });
  if (free.length) parts.push('_' + optQuartersLabelEn_(free) + ' open_');
  return parts.join(' · ');
}

/* ---------------- 1분 작업: 캘린더 동기화 + 하드웨어 알림 ---------------- */
function minuteJob() {

/* rev 가 그대로여도 이 간격마다 한 번은 전체 재동기화한다.
 * (클라이언트의 rev 증가가 실패했거나, 누군가 구글캘린더를 직접 손댄 경우의 자가 치유) */
const SAFETY_RESYNC_MIN = 10;

function calendarSync_(force) {
  if (!GAS_CONFIG.calendarId) return;
  const token = anonToken_();
  const props = PropertiesService.getScriptProperties();

  // ── 1단계: 변경 감지 (문서 1건만 읽음) ───────────────────────────────
  // 예전에는 매분 예약·스케줄 전체를 다시 읽어 해시를 만들었고, 그 읽기량이
  // Firestore 무료 한도(5만/일)를 넘겨 앱 전체가 permission/quota 오류를 낸 이력이 있다.
  const meta = fsGet_(token, 'syncMeta/wos');
  const rev = String((meta && meta.rev) != null ? meta.rev : 0);
  const lastAt = parseInt(props.getProperty('wos_sync_at') || '0', 10);
  const stale = (Date.now() - lastAt) >= SAFETY_RESYNC_MIN * 60000;
  if (!force && !stale && props.getProperty('wos_sync_rev') === rev) return; // 평소엔 여기서 종료 = 1 read

  // ── 2단계: 실제 변경이 있을 때만 전체 조회 ──────────────────────────
  const c = phaseOf(Date.now());
  const states = [c.votingWeek, c.targetWeek].map(function (w) { return buildDesired_(token, w); });
  const hash = sha256Hex(JSON.stringify(states));
  // 본 시점의 rev 를 저장 (동기화 도중 들어온 변경은 rev 가 다시 올라 다음 분에 처리됨)
  props.setProperty('wos_sync_rev', rev);
  props.setProperty('wos_sync_at', String(Date.now()));
  if (!force && !stale && props.getProperty('wos_cal_hash') === hash) return;
  const memberDocs = fsList_(token, 'members');
  const names = {}; memberDocs.forEach(function (m) { names[m.id] = m.name; });
  states.forEach(function (s) { if (s) reconcileWeek_(s.week, s.blocks, names); });
  props.setProperty('wos_cal_hash', hash);
}

/* 주간 유효 스케줄 → 연속 블록 목록 (결정적 직렬화) */
function buildDesired_(token, week) {
  const admDocs = fsQueryByWeek_(token, 'opticsAdminBlocks', week);
  const adm = {};
  admDocs.forEach(function (d) { adm[String(d.day)] = d; });
  const results = fsGet_(token, 'opticsResults/' + week);
  // 배정 발표 전이라도 관리자 지정 일정(Repair·외부인 방문측정)은 캘린더에 올린다
  if (!results && !admDocs.length) return null;
  const schedDocs = fsQueryByWeek_(token, 'opticsSchedule', week);
  const sched = {};
  schedDocs.forEach(function (d) { sched[String(d.day)] = d; });
  // ★ 결과 문서 전체를 넘긴다. dayAssign(대표 1명)만 넘기면 그 사람이 하루 전체를
  //   가진 것으로 계산돼, 같은 날 다른 시간대 당첨자의 일정이 대표 이름으로 캘린더에
  //   올라간다(2026-10 실사례: 수요일 Q3·Q4 유원정 → 캘린더엔 조형구 17~24시).
  //   웹앱과 반드시 같은 optEffectiveSchedule 을 써야 한다.
  const eff = optEffectiveSchedule(results || {}, sched, adm);
  const blocks = [];
  for (let d = 0; d < 7; d++) {
    optMergeBlocks(eff[d]).forEach(function (b) {
      blocks.push({ day: d, start: b.start, end: b.end, owner: b.owner });
    });
  }
  return { week: week, blocks: blocks };
}

function reconcileWeek_(week, blocks, names) {
  const cal = CalendarApp.getCalendarById(GAS_CONFIG.calendarId);
  if (!cal) throw new Error('캘린더 접근 실패: ' + GAS_CONFIG.calendarId);
  const desired = {}; // key 'WOS|week|day|start|end|owner'
  blocks.forEach(function (b) {
    desired['WOS|' + week + '|' + b.day + '|' + b.start + '|' + b.end + '|' + b.owner] = b;
  });
  const t0 = new Date(weekStartMs(week) - 3600000);
  const t1 = new Date(weekStartMs(week + 1) + 3600000);
  const seen = {};
  cal.getEvents(t0, t1).forEach(function (ev) {
    const m = (ev.getDescription() || '').match(/\[WOS\|(\d+)\|(\d)\|(\d+)\|(\d+)\|([^\]]+)\]/);
    if (!m || parseInt(m[1], 10) !== week) return;
    const key = 'WOS|' + m[1] + '|' + m[2] + '|' + m[3] + '|' + m[4] + '|' + m[5];
    if (!desired[key] || seen[key]) { ev.deleteEvent(); return; }
    seen[key] = true;
    // 제목·설명이 지금 규칙과 다르면 고친다(영어 전환·이름 변경 반영)
    const want = calTitleEn_(desired[key].owner, names);
    if (ev.getTitle() !== want) ev.setTitle(want);
    const wantDesc = CAL_DESC_EN_ + '\n[' + key + ']';
    if (ev.getDescription() !== wantDesc) ev.setDescription(wantDesc);
  });
  let created = 0;
  for (const key in desired) {
    if (seen[key]) continue;
    const b = desired[key];
    const title = calTitleEn_(b.owner, names);
    const desc = CAL_DESC_EN_ + '\n[' + key + ']';
    const base = weekStartMs(week) + b.day * DAY_MS;
    cal.createEvent(title, new Date(base + b.start * 1800000), new Date(base + b.end * 1800000), { description: desc });
    created++;
  }
  Logger.log('캘린더 동기화: week ' + week + ' — 블록 ' + blocks.length + ', 신규 ' + created);
}

/* 하드웨어 변경 로그 → Slack (마지막 공지 이후 신규분만) */
function hardwareNotify_() {
  const props = PropertiesService.getScriptProperties();
  const raw = props.getProperty('wos_hw_last');
  const token = anonToken_();
  // 워터마크가 없으면(최초 실행/속성 초기화) 과거 로그를 한꺼번에 Slack 에 쏟지 않도록 기준만 세운다
  if (raw === null) { props.setProperty('wos_hw_last', String(Date.now())); Logger.log('하드웨어 알림 기준 시각 초기화'); return; }
  const lastTs = parseInt(raw || '0', 10);
  const fresh = fsQueryHwSince_(token, lastTs); // 신규분만 (없으면 1 read)
  if (!fresh.length) return;
  const memberDocs = fsList_(token, 'members');
  const names = {}; memberDocs.forEach(function (m) { names[m.id] = m.name; });
  const lines = ['🛠 *Optics hardware configuration changed*'];
  fresh.forEach(function (l) {
    lines.push('');
    lines.push('*[' + (HW_SECTION_EN_[l.section] || l.section) + ']* — ' + (names[l.memberId] || '?') + ' · ' + optFmtDateDowEn_(l.tsMs || Date.now()));
    lines.push(hwSummaryEn_(l.summary));
  });
  if (GAS_CONFIG.appUrl) { lines.push(''); lines.push('👉 Check the status board: ' + GAS_CONFIG.appUrl); }
  postToSlack_(lines.join('\n'));
  props.setProperty('wos_hw_last', String(fresh[fresh.length - 1].tsMs || Date.now()));
  Logger.log('하드웨어 변경 ' + fresh.length + '건 공지');
}

/* ---------------- 스케줄 관리 (합성 봇과 동일한 2단계 정각 실행) ---------------- */
function armDaily() {
  try { PropertiesService.getScriptProperties().deleteProperty('wos_cal_hash'); } catch (e) {}
  ScriptApp.getProjectTriggers().forEach(function (t) {
    const h = t.getHandlerFunction();
    if (h === 'announceJob' || h === 'remindJob') ScriptApp.deleteTrigger(t);
  });
  const nowT = Date.now();
  const dow = kstDow(nowT); // 0=월 ... 6=일
  const handler = dow <= 2 ? 'remindJob' : (dow === 3 ? 'announceJob' : null);
  if (!handler) { Logger.log('오늘(' + '월화수목금토일'[dow] + ')은 발송 없음'); return; }
  const noonUtcMs = kstDayNum(nowT) * 86400000 - KST_OFFSET_MS + 12 * 3600000 + 20000;
  if (noonUtcMs <= nowT) { Logger.log('이미 12시 경과'); return; }
  ScriptApp.newTrigger(handler).timeBased().at(new Date(noonUtcMs)).create();
  Logger.log(handler + ' 을 오늘 12:00(KST)에 장전 완료');
}
function setup() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('armDaily').timeBased().everyDays(1).atHour(9).create();
  ScriptApp.newTrigger('minuteJob').timeBased().everyMinutes(CAL_SYNC_EVERY_MINUTES).create();
  armDaily();
  Logger.log('설정 완료: 매일 09시대 armDaily(월화수 12:00 리마인드/목 12:00 발표) + ' + CAL_SYNC_EVERY_MINUTES + '분마다 캘린더·하드웨어 알림');
}
