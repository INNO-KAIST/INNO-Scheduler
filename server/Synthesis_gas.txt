/* =====================================================================
 * Weekly Synthesis Scheduler - 핵심 공유 로직 (주차 계산 + 결정론적 추첨)
 * 이 파일은 index.html 에 그대로 인라인되며, Node 단위테스트에서도 사용됨
 * ===================================================================== */

const KST_OFFSET_MS = 9 * 3600 * 1000; // Asia/Seoul, DST 없음
const DAY_MS = 86400000;

/* ---- 슬롯 정의 (canonical order) ----
 * Day   : 월~목 06:00-18:00  → D0~D3
 * Night : 월~목 18:00-익일06 → N0~N3
 * Weekend: 금/토/일 전일     → W0~W2
 * 배치: 1~4  → cellKey = "D0-1" 형태 (총 44개)
 */
const SLOT_DEFS = [
  { key: 'D0', period: 'Day',     label: '월요일 06-18시',        day: '월' },
  { key: 'D1', period: 'Day',     label: '화요일 06-18시',        day: '화' },
  { key: 'D2', period: 'Day',     label: '수요일 06-18시',        day: '수' },
  { key: 'D3', period: 'Day',     label: '목요일 06-18시',        day: '목' },
  { key: 'N0', period: 'Night',   label: '월요일 18시-화 06시',   day: '월' },
  { key: 'N1', period: 'Night',   label: '화요일 18시-수 06시',   day: '화' },
  { key: 'N2', period: 'Night',   label: '수요일 18시-목 06시',   day: '수' },
  { key: 'N3', period: 'Night',   label: '목요일 18시-금 06시',   day: '목' },
  { key: 'W0', period: 'Weekend', label: '금요일 (전일)',          day: '금' },
  { key: 'W1', period: 'Weekend', label: '토요일 (전일)',          day: '토' },
  { key: 'W2', period: 'Weekend', label: '일요일 (전일)',          day: '일' },
];
const SLOT_ORDER = SLOT_DEFS.map(s => s.key);
const BATCHES = [1, 2, 3, 4];
const ALL_CELL_KEYS = SLOT_ORDER.flatMap(s => BATCHES.map(b => `${s}-${b}`));
const TOKENS_PER_WEEK = 4;
const DAY_CAP = 2; // 1주 Day 시간대 1인 최대 배치 수

function isDaySlot(slotKey) { return slotKey[0] === 'D'; }
function parseCell(cellKey) {
  const [slot, b] = cellKey.split('-');
  return { slot, batch: parseInt(b, 10) };
}

/* ---- 주차 계산 (월요일 시작, KST 고정) ---- */
function kstDayNum(ms) { return Math.floor((ms + KST_OFFSET_MS) / DAY_MS); }
function kstDow(ms) { return (kstDayNum(ms) + 3) % 7; } // 0=월 ... 6=일 (1970-01-01=목 검증됨)
// 주 인덱스(월요일 시작). 월요일의 dayNum ≡ 4 (mod 7) 이므로 +3 보정 후 7로 나눠 항상 '정수' 반환
function weekIndexOf(ms) { return Math.floor((kstDayNum(ms) + 3) / 7); }
function weekStartMs(weekIdx) { return (weekIdx * 7 - 3) * DAY_MS - KST_OFFSET_MS; } // 해당 주 월요일 00:00 KST의 UTC ms
function fmtDate(ms) {
  const d = new Date(ms + KST_OFFSET_MS);
  return `${d.getUTCMonth() + 1}/${d.getUTCDate()}`;
}
function weekLabel(weekIdx) {
  const s = weekStartMs(weekIdx);
  return `${fmtDate(s)}(월) ~ ${fmtDate(s + 6 * DAY_MS)}(일)`;
}
/* 투표 주(현재 주)의 국면 판정. targetWeek = votingWeek + 1 */
function phaseOf(nowMs) {
  const votingWeek = weekIndexOf(nowMs);
  const targetWeek = votingWeek + 1;
  const monday = weekStartMs(votingWeek);
  const voteClose = monday + 3 * DAY_MS;            // 목요일 00:00 (= 수요일 24시)
  const announce = monday + 3 * DAY_MS + 12 * 3600 * 1000; // 목요일 12:00
  let phase;
  if (nowMs < voteClose) phase = 'VOTING';
  else if (nowMs < announce) phase = 'WAITING';
  else phase = 'BOOKING';
  return { votingWeek, targetWeek, monday, voteClose, announce, phase };
}

/* ---- 결정론적 시드 및 PRNG ---- */
/* 투표 스냅샷의 canonical JSON: memberId 정렬, 각 멤버의 cell 목록 정렬 */
function canonicalVotesJson(votesByMember) {
  const ids = Object.keys(votesByMember).sort();
  const obj = {};
  for (const id of ids) {
    obj[id] = [...votesByMember[id]].sort();
  }
  return JSON.stringify(obj);
}

/* SHA-256 (의존성 없는 순수 JS 구현; 브라우저/Node 동일 결과) */
function sha256Hex(str) {
  const K = [0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,
    0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,
    0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,
    0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,
    0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,
    0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,0xd192e819,0xd6990624,0xf40e3585,0x106aa070,
    0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,
    0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2];
  let H = [0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19];
  const rr = (x, n) => (x >>> n) | (x << (32 - n));
  // UTF-8 인코딩
  const bytes = [];
  for (let i = 0; i < str.length; i++) {
    let c = str.codePointAt(i);
    if (c > 0xffff) i++;
    if (c < 0x80) bytes.push(c);
    else if (c < 0x800) bytes.push(0xc0 | (c >> 6), 0x80 | (c & 63));
    else if (c < 0x10000) bytes.push(0xe0 | (c >> 12), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
    else bytes.push(0xf0 | (c >> 18), 0x80 | ((c >> 12) & 63), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
  }
  const bitLen = bytes.length * 8;
  bytes.push(0x80);
  while (bytes.length % 64 !== 56) bytes.push(0);
  for (let i = 7; i >= 0; i--) bytes.push((i >= 4 ? 0 : (bitLen / Math.pow(2, i * 8))) & 0xff);
  // 상위 32비트는 실용상 0 (2^32비트 미만 입력)
  for (let off = 0; off < bytes.length; off += 64) {
    const w = new Array(64);
    for (let i = 0; i < 16; i++)
      w[i] = (bytes[off + 4 * i] << 24) | (bytes[off + 4 * i + 1] << 16) | (bytes[off + 4 * i + 2] << 8) | bytes[off + 4 * i + 3];
    for (let i = 16; i < 64; i++) {
      const s0 = rr(w[i - 15], 7) ^ rr(w[i - 15], 18) ^ (w[i - 15] >>> 3);
      const s1 = rr(w[i - 2], 17) ^ rr(w[i - 2], 19) ^ (w[i - 2] >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) | 0;
    }
    let [a, b, c, d, e, f, g, h] = H;
    for (let i = 0; i < 64; i++) {
      const S1 = rr(e, 6) ^ rr(e, 11) ^ rr(e, 25);
      const ch = (e & f) ^ (~e & g);
      const t1 = (h + S1 + ch + K[i] + w[i]) | 0;
      const S0 = rr(a, 2) ^ rr(a, 13) ^ rr(a, 22);
      const mj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (S0 + mj) | 0;
      h = g; g = f; f = e; e = (d + t1) | 0; d = c; c = b; b = a; a = (t1 + t2) | 0;
    }
    H = H.map((x, i) => (x + [a, b, c, d, e, f, g, h][i]) | 0);
  }
  return H.map(x => (x >>> 0).toString(16).padStart(8, '0')).join('');
}

/* mulberry32 - 시드 기반 결정론적 PRNG */
function makeRng(seedHex) {
  let s = parseInt(seedHex.slice(0, 8), 16) >>> 0;
  // 시드 확장: 해시 뒷부분도 섞음
  s = (s ^ parseInt(seedHex.slice(8, 16), 16)) >>> 0;
  return function () {
    s = (s + 0x6D2B79F5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function shuffled(arr, rng) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

/* =====================================================================
 * 결정론적 추첨
 * 입력: votesByMember = { memberId: ["D0-1","D0-2","N1-3", ...] }
 * 출력: { seed, assignments: {cellKey: memberId}, priorityMembers: [...],
 *         dayCount: {memberId: n} }
 *
 * 규칙 구현:
 * (1) 동일 슬롯+배치 1인 배정, 경합 시 시드 기반 무작위 추첨
 * (2) Day 시간대 주간 1인 최대 2배치 (배정 시점 강제)
 * (3) 동일 슬롯에 2개 이상 배치 투표자(번들)가 우선권, 번들 간 경합도 무작위
 * (4) 번들은 all-or-nothing (전부 배정 또는 전부 탈락)
 * ===================================================================== */
function computeLottery(votesByMember) {
  const canonical = canonicalVotesJson(votesByMember);
  const seed = sha256Hex(canonical);
  const rng = makeRng(seed);

  const assignments = {}; // cellKey -> memberId
  const dayCount = {};    // memberId -> Day 배정 수
  const memberIds = Object.keys(votesByMember).sort();

  for (const slot of SLOT_ORDER) {
    const day = isDaySlot(slot);
    // 이 슬롯에서 멤버별 투표 배치 목록
    const bySlot = [];
    for (const m of memberIds) {
      const bs = (votesByMember[m] || [])
        .map(parseCell).filter(c => c.slot === slot).map(c => c.batch)
        .sort((a, b) => a - b);
      if (bs.length) bySlot.push({ m, bs });
    }
    // --- 1단계: 번들(같은 슬롯 2개 이상 배치) 우선, 무작위 순서 그리디 ---
    const bundles = shuffled(bySlot.filter(x => x.bs.length >= 2), rng);
    for (const { m, bs } of bundles) {
      if (day && (dayCount[m] || 0) + bs.length > DAY_CAP) continue; // 규칙(2)
      const free = bs.every(b => !assignments[`${slot}-${b}`]);
      if (free) { // 규칙(4): all-or-nothing
        for (const b of bs) assignments[`${slot}-${b}`] = m;
        if (day) dayCount[m] = (dayCount[m] || 0) + bs.length;
      }
    }
    // --- 2단계: 단일 투표자, 배치별 무작위 추첨 ---
    const singles = bySlot.filter(x => x.bs.length === 1);
    for (const b of BATCHES) {
      const key = `${slot}-${b}`;
      if (assignments[key]) continue;
      const cands = singles
        .filter(x => x.bs[0] === b)
        .filter(x => !(day && (dayCount[x.m] || 0) >= DAY_CAP))
        .map(x => x.m); // memberIds 순회로 이미 정렬됨
      if (!cands.length) continue;
      const pick = cands.length === 1 ? cands[0] : cands[Math.floor(rng() * cands.length)];
      assignments[key] = pick;
      if (day) dayCount[pick] = (dayCount[pick] || 0) + 1;
    }
  }

  // Day에 1표 이상 투자했으나 Day 배정 0인 인원 → 우선예약 대상
  const priorityMembers = memberIds.filter(m => {
    const votedDay = (votesByMember[m] || []).some(c => isDaySlot(parseCell(c).slot));
    return votedDay && !(dayCount[m] > 0);
  });

  return { seed, assignments, priorityMembers, dayCount, canonical };
}

/* 투표 유효성 검사 (클라이언트 측; 서버 rules 와 동일 기준) */
function validateVoteSet(cells) {
  if (!Array.isArray(cells)) return '형식 오류';
  if (cells.length > TOKENS_PER_WEEK) return `토큰은 주당 ${TOKENS_PER_WEEK}개까지입니다.`;
  const set = new Set(cells);
  if (set.size !== cells.length) return '동일 슬롯·배치에 중복 투표할 수 없습니다.';
  for (const c of cells) if (!ALL_CELL_KEYS.includes(c)) return `유효하지 않은 슬롯: ${c}`;
  // 단일 Day 슬롯 내 3개 이상 번들은 규칙(2)와 (4)의 조합상 절대 당첨 불가 → 사전 차단
  for (const s of SLOT_ORDER.filter(isDaySlot)) {
    const n = cells.filter(c => parseCell(c).slot === s).length;
    if (n > DAY_CAP) return `Day 슬롯(${s}) 하나에 ${n}개 배치 묶음은 Day 주간 2개 제한과 전부-또는-전무 규칙상 당첨이 불가능하여 차단됩니다.`;
  }
  return null;
}

function http_(url, opt) {
  const resp = UrlFetchApp.fetch(url, Object.assign({ muteHttpExceptions: true }, opt || {}));

  let j = null;

}

function fromFs_(v) {
  if (v.stringValue !== undefined) return v.stringValue;
  if (v.integerValue !== undefined) return parseInt(v.integerValue, 10);
  if (v.doubleValue !== undefined) return v.doubleValue;
  if (v.booleanValue !== undefined) return v.booleanValue;
  if (v.nullValue !== undefined) return null;
  if (v.timestampValue !== undefined) return v.timestampValue;
  if (v.arrayValue !== undefined) return (v.arrayValue.values || []).map(fromFs_);
  if (v.mapValue !== undefined) return decodeFields_(v.mapValue.fields || {});
  return null;
}
function decodeFields_(fields) {
  const o = {};
  for (const k in fields) o[k] = fromFs_(fields[k]);
  return o;
}
function toFs_(x) {
  if (x === null) return { nullValue: null };
  if (typeof x === 'string') return { stringValue: x };
  if (typeof x === 'boolean') return { booleanValue: x };
  if (typeof x === 'number') return Number.isInteger(x) ? { integerValue: String(x) } : { doubleValue: x };
  if (Array.isArray(x)) return { arrayValue: { values: x.map(toFs_) } };
  if (typeof x === 'object') {
    const fields = {};
    for (const k in x) fields[k] = toFs_(x[k]);
    return { mapValue: { fields: fields } };
  }
  throw new Error('encode 불가: ' + typeof x);
}
function encodeFields_(obj) {
  const fields = {};
  for (const k in obj) fields[k] = toFs_(obj[k]);
  return fields;
}
function fsGet_(token, path) {
  const r = http_(FS_BASE_() + '/' + path, { headers: { Authorization: 'Bearer ' + token } });
  if (r.code === 404) return null;
  if (r.j && r.j.error) throw new Error('GET ' + path + ': ' + r.j.error.message);
  return decodeFields_((r.j && r.j.fields) || {});
}
function fsList_(token, collection) {
  let docs = [], pageToken = '';
  do {
    const r = http_(FS_BASE_() + '/' + collection + '?pageSize=300' + (pageToken ? '&pageToken=' + pageToken : ''),
      { headers: { Authorization: 'Bearer ' + token } });
    if (r.j && r.j.error) throw new Error('LIST ' + collection + ': ' + r.j.error.message);
    ((r.j && r.j.documents) || []).forEach(function (d) {
      docs.push(Object.assign({ id: d.name.split('/').pop() }, decodeFields_(d.fields || {})));
    });
    pageToken = (r.j && r.j.nextPageToken) || '';
  } while (pageToken);
  return docs;
}
function fsQueryByWeek_(token, collection, week) {
  const r = http_(FS_BASE_() + ':runQuery', {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ structuredQuery: {
      from: [{ collectionId: collection }],
      where: { fieldFilter: { field: { fieldPath: 'week' }, op: 'EQUAL', value: { integerValue: String(week) } } },
    } }),
  });
  if (r.j && r.j.error) throw new Error('QUERY ' + collection + ': ' + r.j.error.message);
  return (Array.isArray(r.j) ? r.j : []).filter(function (x) { return x.document; })
    .map(function (x) { return Object.assign({ id: x.document.name.split('/').pop() }, decodeFields_(x.document.fields || {})); });
}
/* 문서 1건 생성-또는-갱신(PATCH + updateMask). 없으면 만들고 있으면 해당 필드만 덮어쓴다. */
function fsSet_(token, path, obj) {
  const mask = Object.keys(obj).map(function (k) { return 'updateMask.fieldPaths=' + encodeURIComponent(k); }).join('&');
  const r = http_(FS_BASE_() + '/' + path + '?' + mask, {
    method: 'patch', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ fields: encodeFields_(obj) }),
  });
  if (r.j && r.j.error) throw new Error('SET ' + path + ': ' + r.j.error.message);
  return true;
}

/* ── 하트비트 ───────────────────────────────────────────────────────────
 * 정상 동작할 때마다 syncMeta/wss_bot.okMs 를 갱신한다. 웹앱은 이 값이
 * 60분 이상 낡으면 화면 상단에 "구글 캘린더 동기화가 멈춰 있습니다" 배너를 띄운다.
 * 30분 기록 + 60분 경고 = "한 번 걸러도 괜찮고, 두 번 연속 놓치면 경고" 구조.
 *
 * 왜 필요한가 — 2026-10-02, 이 GAS 프로젝트가 휴지통에 들어가면서 5일간 멈췄다.
 * 휴지통에 있는 스크립트는 Google 이 아예 호출하지 않으므로 실행 오류도,
 * 실패 요약 메일도 남지 않는다. 트리거는 '등록됨 · 오류율 0%' 로 멀쩡해 보였고
 * 아무도 알아채지 못한 채 구글 캘린더만 조용히 낡아갔다. 이 하트비트는 그
 * '조용한 죽음'을 사람이 볼 수 있게 만드는 장치다.
 *
 * 쓰기 비용: 1분마다가 아니라 최대 30분 간격 = 봇당 하루 약 48회. 무료 쓰기 한도
 * (2만/일)에 비하면 무시할 수준이다. 실패해도 본 작업을 막지 않는다. */
const HEARTBEAT_MIN = 30;

function heartbeat_() {
  try {
    const props = PropertiesService.getScriptProperties();
    const now = Date.now();
    const last = parseInt(props.getProperty('wss_hb_at') || '0', 10);
    if (now - last < HEARTBEAT_MIN * 60000) return;
    fsSet_(anonToken_(), 'syncMeta/wss_bot', { okMs: now });
    props.setProperty('wss_hb_at', String(now));
  } catch (e) {
    Logger.log('하트비트 기록 실패(무시됨): ' + e.message);
  }
}

function fsCreate_(token, collection, docId, obj) {
  const r = http_(FS_BASE_() + '/' + collection + '?documentId=' + encodeURIComponent(docId), {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ fields: encodeFields_(obj) }),
  });
  if (r.j && r.j.error) {
    if (r.code === 409) return false; // 이미 존재 (다른 곳에서 생성됨 - 정상)
    throw new Error('CREATE ' + collection + '/' + docId + ': ' + r.j.error.message);
  }
  return true;
}

function buildAnnounceMessage_(targetWeek, results, bookings, members, config) {
  const nameOf = function (id) { return (members[id] && members[id].name) || '(알 수 없음)'; };
  const a = results.assignments || {};
  const lines = [];
  lines.push('📢 *합성 스케줄 확정 발표 — ' + weekLabel(targetWeek) + '*');
  lines.push('');
  const byPeriod = { Day: [], Night: [], Weekend: [] };
  SLOT_DEFS.forEach(function (sd) {
    BATCHES.forEach(function (b) {
      const key = sd.key + '-' + b;
      if (a[key]) byPeriod[sd.period].push('  • ' + sd.label + ' · Batch #' + b + ' — *' + nameOf(a[key]) + '*');
      else if (bookings[key]) byPeriod[sd.period].push('  • ' + sd.label + ' · Batch #' + b + ' — *' + nameOf(bookings[key].memberId) + '* (예약)');
    });
  });
  const nAssigned = Object.keys(a).length, nBooked = Object.keys(bookings).length;
  lines.push('🎲 *추첨 확정: ' + nAssigned + '건*' + (nBooked ? ' / 예약: ' + nBooked + '건' : ''));
  ['Day', 'Night', 'Weekend'].forEach(function (p) {
    if (byPeriod[p].length) { lines.push('*[' + p + ']*'); byPeriod[p].forEach(function (l) { lines.push(l); }); }
  });
  if (!nAssigned && !nBooked) lines.push('_이번 주 배정 없음 (투표 없음)_');
  lines.push('');
  const prio = results.priorityMembers || [];
  const prioMin = (config && config.priorityWindowMin != null) ? config.priorityWindowMin : 60;
  if (prio.length && prioMin > 0) {
    lines.push('🎗 *Day 우선예약 대상* (Day 지원했으나 미선정): ' + prio.map(function (id) { return mention_(nameOf(id)); }).join(', '));
    lines.push('   → 지금부터 *' + prioMin + '분간* 위 인원만 빈 슬롯 선착순 예약 가능, 이후 전체 오픈');
  } else {
    lines.push('⚡ 빈 슬롯 실시간 예약이 *지금부터* 전체 오픈되었습니다 (1인 횟수 제한 없음)');
  }
  lines.push('▫ 남은 빈 슬롯: *' + (44 - nAssigned - nBooked) + '칸*');
  if (GAS_CONFIG.appUrl) { lines.push(''); lines.push('👉 예약 바로가기: ' + GAS_CONFIG.appUrl); }
  lines.push('_추첨 시드(투표 데이터 해시): ' + String(results.seed || '').slice(0, 16) + '… — 앱 «투표 결과» 탭에서 누구나 검증 가능_');
  return lines.join('\n');
}

function buildRemindMessage_(targetWeek, voteClose, activeMembers, votesByMember) {
  const nowMs = Date.now();
  const voted = activeMembers.filter(function (m) { return (votesByMember[m.id] || []).length > 0; });
  // 멘션 대상: Slack 채널 멤버(SLACK_MENTION 등재자) 중 아직 토큰을 1개도 쓰지 않은 인원만
  const notVoted = activeMembers.filter(function (m) {
    return (votesByMember[m.id] || []).length === 0 && SLACK_MENTION[m.name];
  });
  const leftMs = voteClose - nowMs;
  const leftH = Math.floor(leftMs / 3600000);
  const leftStr = leftH >= 24 ? Math.floor(leftH / 24) + '일 ' + (leftH % 24) + '시간' : leftH + '시간 ' + Math.floor(leftMs % 3600000 / 60000) + '분';
  const isLastDay = leftMs <= 12.5 * 3600000;

  const lines = [];
  lines.push('📮 *합성 스케줄 사전 투표(토큰) 리마인드 — 대상: ' + weekLabel(targetWeek) + '*');
  lines.push('');
  lines.push(isLastDay ? '⚠ *오늘 자정(수 24:00) 마감입니다!*' : '투표 마감까지 *' + leftStr + '* (수요일 24:00 KST 마감)');
  lines.push('✅ 투표 완료: *' + voted.length + '명* / 전체 ' + activeMembers.length + '명');
  if (notVoted.length) {
    lines.push('⏳ 아직 투표하지 않은 분들: ' + notVoted.map(function (m) { return mention_(m.name); }).join(' '));
  }
  lines.push('');
  lines.push('• 1인당 토큰 4개 · 같은 시간표에 2개 이상 배치 시 우선권(묶음 처리) · 결과 발표: 목 12:00');
  if (GAS_CONFIG.appUrl) lines.push('👉 투표하러 가기: ' + GAS_CONFIG.appUrl);
  return lines.join('\n');
}

/* ---------------- 실행 작업 ---------------- */
/* 월/화/수 12:00 — 투표 리마인드 (미투표자만 멘션, 전원 완료 시 발송 생략) */
function remindJob() {
  const c = phaseOf(Date.now());
  if (c.phase !== 'VOTING') { Logger.log('투표 기간이 아니므로 리마인드 생략'); return; }
  const token = anonToken_();
  const memberDocs = fsList_(token, 'members');
  const activeMembers = memberDocs.filter(function (m) { return m.active !== false; });
  const voteDocs = fsQueryByWeek_(token, 'votes', c.targetWeek);
  const votesByMember = {};
  voteDocs.forEach(function (d) { votesByMember[d.memberId] = Object.keys(d.cells || {}); });
  // 채널 멤버(SLACK_MENTION 등재자) 중 미투표자가 있을 때만 발송
  const notVoted = activeMembers.filter(function (m) {
    return (votesByMember[m.id] || []).length === 0 && SLACK_MENTION[m.name];
  });
  if (!notVoted.length) { Logger.log('채널 멤버 전원 투표 완료 — 리마인드 생략'); return; }
  postToSlack_(buildRemindMessage_(c.targetWeek, c.voteClose, activeMembers, votesByMember));
  Logger.log('리마인드 발송 완료 (채널 내 미투표 ' + notVoted.length + '명)');
}

/* 목 12:00 — 추첨 결과 확정 + 발표 */
function announceJob() {
  const nowMs = Date.now();
  const c = phaseOf(nowMs);
  if (nowMs < c.announce) { Logger.log('아직 발표 시각 전'); return; }
  const token = anonToken_();

  let results = fsGet_(token, 'results/' + c.targetWeek);
  if (!results) {
    // 결정론적 추첨을 직접 수행하여 게시 (앱과 동일 알고리즘 → 동일 결과 보장)
    const voteDocs = fsQueryByWeek_(token, 'votes', c.targetWeek);
    const votesByMember = {};
    voteDocs.forEach(function (d) { votesByMember[d.memberId] = Object.keys(d.cells || {}).sort(); });
    const r = computeLottery(votesByMember);
    fsCreate_(token, 'results', String(c.targetWeek), {
      week: c.targetWeek, seed: r.seed, assignments: r.assignments,
      priorityMembers: r.priorityMembers, votesJson: r.canonical, createdBy: 'gas-bot',
    });
    results = fsGet_(token, 'results/' + c.targetWeek);
  }
  const memberDocs = fsList_(token, 'members');
  const members = {};
  memberDocs.forEach(function (m) { members[m.id] = m; });
  const bookingDocs = fsQueryByWeek_(token, 'bookings', c.targetWeek);
  const bookings = {};
  bookingDocs.forEach(function (b) { bookings[b.cellKey] = b; });
  const config = fsGet_(token, 'config/app');
  // 확정 취소된 칸은 발표 목록에서 제외 (12시 정각에는 보통 없음)
  const cancelDocs = fsQueryByWeek_(token, 'cancellations', c.targetWeek);
  const effAssign = {};
  for (const k in (results.assignments || {})) effAssign[k] = results.assignments[k];
  cancelDocs.forEach(function (x) { delete effAssign[x.cellKey]; });
  const effResults = Object.assign({}, results, { assignments: effAssign });

  postToSlack_(buildAnnounceMessage_(c.targetWeek, effResults, bookings, members, config));
  Logger.log('발표 완료');

  // 발표 직후 캘린더 즉시 동기화 (실패해도 발표에는 영향 없음)
  try { calendarSyncJob(); } catch (e) { Logger.log('캘린더 동기화 오류: ' + e.message); }
}

/* =====================================================================
 * 구글 캘린더 동기화
 * - 확정 스케줄(추첨 배정 + 선착순 예약)을 공용 캘린더에 일정으로 등록
 * - CAL_SYNC_EVERY_MINUTES 간격 재동기화: 새 예약 추가 / 취소 삭제 / 변경 반영
 * - 어떤 주의 스케줄은 [전주 목 12:00 발표 ~ 해당 주 일요일 24:00]까지 계속 관리됨
 *   (매 실행 시 '진행 중인 주 + 발표된 다음 주' 두 주를 동기화. 지난 주 일정은 기록으로 보존)
 * - 봇이 만든 일정은 설명란의 [WSS|주차|슬롯|멤버ID] 태그로 식별하므로
 *   수동으로 만든 다른 일정은 절대 건드리지 않음
 * - 최적화: 확정 상태의 해시를 저장해 두고, 변화가 없는 회차는 캘린더를
 *   건드리지 않고 즉시 종료 (1분 간격에도 무료 실행시간 한도 내 유지)
 * ===================================================================== */
const CAL_SYNC_EVERY_MINUTES = 1; // 1, 5, 10, 15, 30 중 하나. GAS 실행시간 한도 경고 시 5로 올릴 것

const SLOT_CAL_ = {
  D0: { day: 0, type: 'day' }, D1: { day: 1, type: 'day' }, D2: { day: 2, type: 'day' }, D3: { day: 3, type: 'day' },
  N0: { day: 0, type: 'night' }, N1: { day: 1, type: 'night' }, N2: { day: 2, type: 'night' }, N3: { day: 3, type: 'night' },
  W0: { day: 4, type: 'weekend' }, W1: { day: 5, type: 'weekend' }, W2: { day: 6, type: 'weekend' },
};

function calendarSyncJob() { calendarSync_(false); heartbeat_(); }
// calendarSync_ 가 예외를 던지면 heartbeat_ 에 닿지 않는다 = 배너가 뜬다 (의도된 동작)
/* 캘린더에서 봇 일정을 실수로 지웠을 때 등 강제 전체 재동기화 */
function forceCalendarResync() { calendarSync_(true); }

/* rev 가 그대로여도 이 간격마다 한 번은 전체 재동기화한다.
 * (클라이언트의 rev 증가가 실패했거나, 누군가 구글캘린더를 직접 손댄 경우의 자가 치유) */
const SAFETY_RESYNC_MIN = 10;

function calendarSync_(force) {
  if (!GAS_CONFIG.calendarId) { Logger.log('calendarId 미설정 — 캘린더 동기화 생략'); return; }
  const token = anonToken_();
  const props = PropertiesService.getScriptProperties();

  // ── 1단계: 변경 감지 (문서 1건만 읽음) ───────────────────────────────
  // 예전에는 매분 예약·취소 전체를 다시 읽어 해시를 만들었고, 그 읽기량이
  // Firestore 무료 한도(5만/일)를 넘겨 앱 전체가 quota 오류를 낸 이력이 있다.
  const meta = fsGet_(token, 'syncMeta/wss');
  const rev = String((meta && meta.rev) != null ? meta.rev : 0);
  const lastAt = parseInt(props.getProperty('wss_sync_at') || '0', 10);
  const stale = (Date.now() - lastAt) >= SAFETY_RESYNC_MIN * 60000;
  if (!force && !stale && props.getProperty('wss_sync_rev') === rev) return; // 평소엔 여기서 종료 = 1 read

  // ── 2단계: 실제 변경이 있을 때만 전체 조회 ──────────────────────────
  const c = phaseOf(Date.now());
  // 진행 중인 주(이번 주 스케줄)와 다음 주(발표된 경우) 모두 대상
  const states = [c.votingWeek, c.targetWeek].map(function (w) { return buildDesired_(token, w); });
  // 변화 감지: 확정 상태 전체의 해시가 같으면 캘린더 작업 생략
  const hash = sha256Hex(JSON.stringify(states));
  // 본 시점의 rev 를 저장 (동기화 도중 들어온 변경은 rev 가 다시 올라 다음 분에 처리됨)
  props.setProperty('wss_sync_rev', rev);
  props.setProperty('wss_sync_at', String(Date.now()));
  if (!force && !stale && props.getProperty('wss_cal_hash') === hash) return;
  const memberDocs = fsList_(token, 'members');
  const names = {};
  memberDocs.forEach(function (m) { names[m.id] = m.name; });
  states.forEach(function (s) { if (s) reconcileWeek_(s.week, s.desired, names); });
  props.setProperty('wss_cal_hash', hash);
}

/* 해당 주의 '원하는 캘린더 상태' 수집 (결정적 순서로 구성해 해시 안정성 보장) */
function buildDesired_(token, week) {
  const results = fsGet_(token, 'results/' + week);
  if (!results) return null; // 아직 발표되지 않은 주는 건드리지 않음
  const bookingDocs = fsQueryByWeek_(token, 'bookings', week);
  const cancelDocs = fsQueryByWeek_(token, 'cancellations', week);
  const cancelled = {};
  cancelDocs.forEach(function (x) { cancelled[x.cellKey] = true; });
  const desired = {}; // key 'WSS|week|slot|ownerId' -> {slot, owner, batches[]}
  function addCell(cellKey, ownerId) {
    const p = parseCell(cellKey);
    const key = 'WSS|' + week + '|' + p.slot + '|' + ownerId;
    if (!desired[key]) desired[key] = { slot: p.slot, owner: ownerId, batches: [] };
    desired[key].batches.push(p.batch);
  }
  const a = results.assignments || {};
  Object.keys(a).sort().forEach(function (cellKey) { if (!cancelled[cellKey]) addCell(cellKey, a[cellKey]); });
  // 웹앱 cellState() 와 같은 우선순위: 취소되지 않은 추첨 칸은 추첨 당첨자가 이기고,
  // 그 칸의 예약은 무시한다. (관리자가 취소 기록을 지운 뒤에 남은 예약이 있으면
  // 앱은 당첨자만 보여주는데 캘린더엔 두 사람이 겹쳐 올라가던 엣지 케이스 방지)
  bookingDocs.filter(function (b) { return !(a[b.cellKey] && !cancelled[b.cellKey]); })
    .map(function (b) { return b.cellKey + '|' + b.memberId; }).sort()
    .forEach(function (s) { const q = s.split('|'); addCell(q[0], q[1]); });
  for (const k in desired) desired[k].batches.sort(function (x, y) { return x - y; });
  const orderedKeys = Object.keys(desired).sort();
  const ordered = {};
  orderedKeys.forEach(function (k) { ordered[k] = desired[k]; });
  return { week: week, desired: ordered };
}

function reconcileWeek_(week, desired, names) {
  function titleOf(d) {
    return '[합성 #' + d.batches.join('·#') + '] ' + (names[d.owner] || '(알 수 없음)');
  }
  const cal = CalendarApp.getCalendarById(GAS_CONFIG.calendarId);
  if (!cal) throw new Error('캘린더 접근 실패: ' + GAS_CONFIG.calendarId + ' (ID와 편집 권한 확인)');

  // 이 주에 봇이 만든 기존 일정 수집 (목요일 Night는 금 06시에 끝나므로 여유 포함)
  const t0 = new Date(weekStartMs(week) - 3600000);
  const t1 = new Date(weekStartMs(week + 1) + 10 * 3600000);
  const seen = {};
  cal.getEvents(t0, t1).forEach(function (ev) {
    const m = (ev.getDescription() || '').match(/\[WSS\|(\d+)\|([DNW]\d)\|([^\]]+)\]/);
    if (!m || parseInt(m[1], 10) !== week) return; // 봇 태그 없는 일반 일정은 무시
    const key = 'WSS|' + m[1] + '|' + m[2] + '|' + m[3];
    if (!desired[key] || seen[key]) { ev.deleteEvent(); return; } // 취소됨/중복 → 삭제
    seen[key] = true;
    const want = titleOf(desired[key]);
    if (ev.getTitle() !== want) ev.setTitle(want); // 배치 구성 변경 반영
  });

  // 아직 캘린더에 없는 확정 건 생성
  let created = 0;
  for (const key in desired) {
    if (seen[key]) continue;
    const d = desired[key];
    const sc = SLOT_CAL_[d.slot];
    const title = titleOf(d);
    const desc = 'Weekly Synthesis Scheduler 자동 등록 일정입니다. Bot에 의해 자동 동기화되므로 직접 수정/삭제를 금합니다. 예약 변경은 웹에서 진행해주세요\n[WSS|' + week + '|' + d.slot + '|' + d.owner + ']';
    const base = weekStartMs(week) + sc.day * DAY_MS;
    if (sc.type === 'weekend') {
      cal.createAllDayEvent(title, new Date(base + 12 * 3600000), { description: desc });
    } else if (sc.type === 'day') {
      cal.createEvent(title, new Date(base + 6 * 3600000), new Date(base + 18 * 3600000), { description: desc });
    } else { // night: 18:00 ~ 익일 06:00
      cal.createEvent(title, new Date(base + 18 * 3600000), new Date(base + 30 * 3600000), { description: desc });
    }
    created++;
  }
  Logger.log('캘린더 동기화: week ' + week + ' — 확정 ' + Object.keys(desired).length + '건, 신규 ' + created + '건');
}

/* ---------------- 스케줄 관리 (정밀 12시 실행) ----------------
 * GAS의 주간/일간 트리거는 "시간대(예: 12~13시)" 정밀도라 정각 실행이 안 됨.
 * → 매일 오전 9시대에 armDaily가 실행되어, 당일 12:00 정각(±1분)의
 *   1회성 트리거를 장전하는 2단계 방식으로 정각 실행을 보장. */
function armDaily() {
  // 하루 1회 캘린더 전체 재점검 유도(누군가 캘린더에서 봇 일정을 직접 지운 경우 복구)
  try { PropertiesService.getScriptProperties().deleteProperty('wss_cal_hash'); } catch (e) {}
  // 이전에 장전된 1회성 트리거 정리
  ScriptApp.getProjectTriggers().forEach(function (t) {
    const h = t.getHandlerFunction();
    if (h === 'remindJob' || h === 'announceJob') ScriptApp.deleteTrigger(t);
  });
  const now = Date.now();
  const dow = kstDow(now); // 0=월 ... 6=일
  const handler = dow <= 2 ? 'remindJob' : (dow === 3 ? 'announceJob' : null);
  if (!handler) { Logger.log('오늘(' + '월화수목금토일'[dow] + ')은 발송 없음'); return; }
  const noonUtcMs = kstDayNum(now) * 86400000 - KST_OFFSET_MS + 12 * 3600000 + 20000; // 오늘 12:00:20 KST
  if (noonUtcMs <= now) { Logger.log('이미 12시 경과 — 생략'); return; }
  ScriptApp.newTrigger(handler).timeBased().at(new Date(noonUtcMs)).create();
  Logger.log(handler + ' 을 오늘 12:00(KST)에 장전 완료');
}

/* 최초 1회 실행: 전체 스케줄 등록 */
function setup() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('armDaily').timeBased().everyDays(1).atHour(9).create(); // 매일 09시대
  if (GAS_CONFIG.calendarId) {
    ScriptApp.newTrigger('calendarSyncJob').timeBased().everyMinutes(CAL_SYNC_EVERY_MINUTES).create();
  }
  armDaily(); // 오늘분 즉시 장전
  Logger.log('설정 완료: 매일 09시대 armDaily(월화수 12:00 리마인드/목 12:00 발표)'
    + (GAS_CONFIG.calendarId ? ' + ' + CAL_SYNC_EVERY_MINUTES + '분마다 캘린더 동기화' : ' (캘린더 동기화 꺼짐)'));
}
